--- Inside Python: multi-chunk files, insert, append ---
import sefs, glob, pathlib, os, shutil
from sefs import meta
U, PW = "alice01", "Passw0rd#1"
sefs.register_user(U, PW, "passwd")                 # 1
chunks = lambda: len(glob.glob("chunks/*"))
pathlib.Path("report").write_bytes(b"A" * 1024 + b"B" * 1024 + b"C" * 500)
sefs.encrypt_file("report", U, PW)                  # 1
chunks()                                            # 3
sefs.file_size("report", U, PW)                     # 2548
sefs.read_from_file("report", 1019, 10, U, PW)      # b'AAAAABBBBB'  across a chunk boundary
before = set(glob.glob("chunks/*"))
sefs.write_to_file("report", 10, b"xyz", U, PW)     # 1  insert
sefs.read_from_file("report", 8, 7, U, PW)          # b'AAxyzAA'
sefs.file_size("report", U, PW)                     # 2551
chunks()                                            # 3
len(before & set(glob.glob("chunks/*")))            # 0  old chunks removed
pathlib.Path("small").write_bytes(b"s" * 512)
sefs.encrypt_file("small", U, PW)                   # 1
sefs.write_to_file("small", 512, b"t" * 200, U, PW) # 1  append
sefs.file_size("small", U, PW)                      # 712
chunks()                                            # 4  (small is still one chunk)

--- Tampering: flip one bit ---
sefs.system_health_check(U, PW)                     # 1
victim = sorted(glob.glob("chunks/*"))[0]
raw = bytearray(open(victim, "rb").read()); raw[100] ^= 1
open(victim, "wb").write(raw)
sefs.system_health_check(U, PW)                     # -1  tampering detected
sefs.file_integrity_check("report", U, PW)          # shows which file was hit (-1)
sefs.file_integrity_check("small", U, PW)           #   (the other returns 1)
raw[100] ^= 1; open(victim, "wb").write(raw)        # undo the flip
sefs.system_health_check(U, PW)                     # 1

--- Tampering: swap two chunks, then delete one ---
a, b, c = [ch.name for ch in meta.load("report")[0].chunks]
os.replace(f"chunks/{a}", "tmpA"); os.replace(f"chunks/{b}", f"chunks/{a}"); os.replace("tmpA", f"chunks/{b}")
sefs.file_integrity_check("report", U, PW)          # -1  swapped chunks detected
sefs.read_from_file("report", 0, 5, U, PW)          # None
os.replace(f"chunks/{a}", "tmpA"); os.replace(f"chunks/{b}", f"chunks/{a}"); os.replace("tmpA", f"chunks/{b}")
sefs.file_integrity_check("report", U, PW)          # 1  swapped back
shutil.move(f"chunks/{c}", "backupC")
sefs.file_integrity_check("report", U, PW)          # -1  missing chunk detected
shutil.move("backupC", f"chunks/{c}")
sefs.system_health_check(U, PW)                     # 1
exit()

--- Inside Python ---
import sefs
P = "passwd"
sefs.register_user("alice01", "Passw0rd#1", P)      #  1
sefs.register_user("alice01", "Passw0rd#1", P)      # -1  duplicate
sefs.register_user("bob", "Passw0rd#1", P)          # -1  username too short
sefs.register_user("carol01", "short", P)           # -1  password too short
sefs.is_user_valid("alice01", P)                    #  1
sefs.match_user("alice01", "Passw0rd#1", P)         #  1
sefs.match_user("alice01", "WrongPass#1", P)        # -1
print(open(P).read())                               # alice01:<64-hex salt>:<40-hex hash>
sefs.change_user_password("alice01", "Passw0rd#1", "NewPassw0rd=", P)  # 1
print(open(P).read())                               # salt and hash both changed
sefs.match_user("alice01", "Passw0rd#1", P)         # -1  old password no longer works
sefs.delete_user("alice01", "NewPassw0rd=", P)      #  1
sefs.is_user_valid("alice01", P)                    # -1
exit()
